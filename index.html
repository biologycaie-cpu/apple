Important Fuel Safety & Sensor Material Advisory

Standard YF-S201 water flow sensors use nylon/polypropylene housings intended for bench testing with water. For real petrol/gasoline auditing, use a fuel-rated aluminum/acetal/stainless turbine flow sensor (e.g., FS300A Fuel Edition or ATEX-certified inline flowmeter) with Viton/FKM seals and an intrinsically safe, spark-free enclosure.

FuelGuard_ESP32_BLE.ino (Arduino C++)

Interrupt-driven pulse counter with GPIO 27 open-circuit detection and JSON BLE notifications. 

/*
 * FuelGuard BLE — ESP32 Liquid Flow & Petrol Pump Auditor Firmware
 * Hardware: ESP32 DevKit + YF-S201 (or fuel-rated Hall turbine flow sensor)
 * Wiring:
 *   - Sensor VCC (Red)    -> ESP32 VIN (5V)
 *   - Sensor GND (Black)  -> ESP32 GND
 *   - Sensor SIG (Yellow) -> ESP32 GPIO 27 via 10k/20k 5V->3.3V voltage divider
 *     (20k pull-down pulls GPIO 27 to 0V when unplugged for open-circuit fault detection)
 */

#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

// BLE Service & Characteristic UUIDs (Matches FuelGuard BLE Web App)
#define SERVICE_UUID        "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define CHARACTERISTIC_UUID "beb5483e-36e1-4688-b7f5-ea07361b26a8"

// Flow Sensor Configuration
const uint8_t FLOW_SENSOR_PIN = 27;
float calibrationFactor = 450.0; // Default: 450 pulses per liter for YF-S201
float viscosityMultiplier = 1.00;

volatile uint32_t totalPulses = 0;
volatile uint32_t intervalPulses = 0;
volatile unsigned long lastSensorHighMs = 0;

unsigned long lastReportTime = 0;
const unsigned long REPORT_INTERVAL_MS = 500; // Broadcast every 500ms
const unsigned long SENSOR_DISCONNECT_TIMEOUT_MS = 400; // 0V open-circuit threshold

BLEServer* pServer = nullptr;
BLECharacteristic* pCharacteristic = nullptr;
bool deviceConnected = false;
bool oldDeviceConnected = false;

// Hardware Interrupt Service Routine (ISR) on rising edge
void IRAM_ATTR onFlowPulse() {
  totalPulses++;
  intervalPulses++;
  lastSensorHighMs = millis();
}

// BLE Server Connection Callbacks
class ServerCallbacks : public BLEServerCallbacks {
  void onConnect(BLEServer* pServer) override {
    deviceConnected = true;
    Serial.println("[BLE] FuelGuard Web Client Connected");
  }
  void onDisconnect(BLEServer* pServer) override {
    deviceConnected = false;
    Serial.println("[BLE] Client Disconnected");
  }
};

// Handle commands from FuelGuard Web App ("RESET" or "CAL:450,1.00")
class CharacteristicCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic* pChar) override {
    String rxValue = pChar->getValue().c_str();
    rxValue.trim();
    if (rxValue.length() > 0) {
      Serial.print("[BLE RX] Command: ");
      Serial.println(rxValue);

      if (rxValue.equalsIgnoreCase("RESET")) {
        noInterrupts();
        totalPulses = 0;
        intervalPulses = 0;
        interrupts();
        Serial.println("[SENSOR] Counters zeroed.");
      } else if (rxValue.startsWith("CAL:")) {
        int commaIdx = rxValue.indexOf(',');
        if (commaIdx > 4) {
          float newK = rxValue.substring(4, commaIdx).toFloat();
          float newVisc = rxValue.substring(commaIdx + 1).toFloat();
          if (newK > 10.0) calibrationFactor = newK;
          if (newVisc > 0.1) viscosityMultiplier = newVisc;
          Serial.printf("[SENSOR] Updated K=%.1f, Visc=%.3f\n", calibrationFactor, viscosityMultiplier);
        }
      }
    }
  }
};

void setup() {
  Serial.begin(115200);
  Serial.println("\nStarting FuelGuard BLE Flow Auditor...");

  // External 20k pull-down holds GPIO 27 LOW (0V) when sensor is unplugged
  pinMode(FLOW_SENSOR_PIN, INPUT);
  if (digitalRead(FLOW_SENSOR_PIN) == HIGH) {
    lastSensorHighMs = millis();
  }
  attachInterrupt(digitalPinToInterrupt(FLOW_SENSOR_PIN), onFlowPulse, RISING);

  // Initialize BLE Device
  BLEDevice::init("FuelGuard-ESP32");
  pServer = BLEDevice::createServer();
  pServer->setCallbacks(new ServerCallbacks());

  BLEService* pService = pServer->createService(SERVICE_UUID);

  pCharacteristic = pService->createCharacteristic(
    CHARACTERISTIC_UUID,
    BLECharacteristic::PROPERTY_READ |
    BLECharacteristic::PROPERTY_NOTIFY |
    BLECharacteristic::PROPERTY_WRITE |
    BLECharacteristic::PROPERTY_WRITE_NR
  );

  pCharacteristic->addDescriptor(new BLE2902());
  pCharacteristic->setCallbacks(new CharacteristicCallbacks());

  pService->start();

  BLEAdvertising* pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->setScanResponse(true);
  pAdvertising->setMinPreferred(0x06);
  BLEDevice::startAdvertising();

  Serial.println("[BLE] Advertising as 'FuelGuard-ESP32'...");
}

void loop() {
  unsigned long now = millis();

  // Check if Hall sensor signal is present (HIGH idle or active pulses)
  if (digitalRead(FLOW_SENSOR_PIN) == HIGH) {
    lastSensorHighMs = now;
  }
  bool sensorConnected = (lastSensorHighMs > 0) && ((now - lastSensorHighMs) < SENSOR_DISCONNECT_TIMEOUT_MS);

  if (now - lastReportTime >= REPORT_INTERVAL_MS) {
    float elapsedSec = (now - lastReportTime) / 1000.0;
    lastReportTime = now;

    // Atomically snapshot pulse counts
    noInterrupts();
    uint32_t pulsesSnapshot = totalPulses;
    uint32_t deltaPulses = intervalPulses;
    intervalPulses = 0;
    interrupts();

    // Calculate Flow Rate (L/min) & Total Dispensed Volume (Liters)
    float pulseFreqHz = sensorConnected ? (deltaPulses / elapsedSec) : 0.0;
    float flowRateLPM = sensorConnected ? (((pulseFreqHz * 60.0) / calibrationFactor) * viscosityMultiplier) : 0.0;
    float totalLiters = (pulsesSnapshot / calibrationFactor) * viscosityMultiplier;

    // Format JSON telemetry packet with "sensor" health boolean
    char jsonPayload[144];
    snprintf(
      jsonPayload,
      sizeof(jsonPayload),
      "{\"flow\":%.2f,\"liters\":%.3f,\"pulses\":%lu,\"sensor\":%s}",
      flowRateLPM,
      totalLiters,
      (unsigned long)pulsesSnapshot,
      sensorConnected ? "true" : "false"
    );

    Serial.println(jsonPayload);

    if (deviceConnected) {
      pCharacteristic->setValue((uint8_t*)jsonPayload, strlen(jsonPayload));
      pCharacteristic->notify();
    }
  }

  // Re-advertise automatically after disconnect
  if (!deviceConnected && oldDeviceConnected) {
    delay(300);
    pServer->startAdvertising();
    Serial.println("[BLE] Restarted advertising...");
    oldDeviceConnected = deviceConnected;
  }
  if (deviceConnected && !oldDeviceConnected) {
    oldDeviceConnected = deviceConnected;
  }
}
